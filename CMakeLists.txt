cmake_minimum_required(VERSION 3.24)
project(OpenDesertVR VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)
include(FetchContent)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(nlohmann_json URL https://codeload.github.com/nlohmann/json/zip/9cca280a4d0ccf0c08f47a99aa71d1b0e52f8d03 URL_HASH SHA256=42013e884a8a38d2a974619844a294de614dab4bc14e0ad23711ef4dacbb24c1)
FetchContent_MakeAvailable(nlohmann_json)
add_library(odv_math STATIC src/math/stereo.cpp)
target_include_directories(odv_math PUBLIC include)
add_library(odv_core STATIC src/core/core.cpp src/core/metrics.cpp)
target_include_directories(odv_core PUBLIC include)
target_link_libraries(odv_core PUBLIC nlohmann_json::nlohmann_json)
add_library(odv_stereo STATIC src/stereo/scene.cpp)
target_link_libraries(odv_stereo PUBLIC odv_math)
add_library(odv_game_crimson STATIC src/game/crimson_desert/adapter.cpp)
target_link_libraries(odv_game_crimson PUBLIC odv_core)
add_executable(odv-cli apps/odv-cli/main.cpp)
target_link_libraries(odv-cli PRIVATE odv_core odv_game_crimson)
set(odv_targets odv_math odv_core odv_stereo odv_game_crimson odv-cli)
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "M0 requires Windows x64")
  endif()
  set(BUILD_API_LAYERS OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(BUILD_CONFORMANCE_TESTS OFF CACHE BOOL "" FORCE)
  set(DYNAMIC_LOADER ON CACHE BOOL "" FORCE)
  FetchContent_Declare(openxr URL https://codeload.github.com/KhronosGroup/OpenXR-SDK/zip/781f2eab3698d653c804ecbd11e0aed47eaad1c6 URL_HASH SHA256=91b35885bcf4e7cc22696c87dafae02b033d1df98a120b1a910912c520350267)
  FetchContent_MakeAvailable(openxr)
  add_library(odv_d3d12 STATIC src/d3d12/renderer.cpp src/d3d12/stereo_stream.cpp src/d3d12/depth_stereo.cpp src/d3d12/depth_preview.cpp)
  target_link_libraries(odv_d3d12 PUBLIC odv_core odv_stereo d3d12 dxgi d3dcompiler)
  add_library(odv_openxr STATIC src/openxr/host.cpp)
  target_link_libraries(odv_openxr PUBLIC odv_d3d12 OpenXR::openxr_loader)
  target_link_libraries(odv-cli PRIVATE odv_openxr)
  add_executable(odv-testhost apps/odv-testhost/main.cpp)
  target_link_libraries(odv-testhost PRIVATE odv_openxr)
  add_executable(OpenDesertVR WIN32 apps/odv-desktop/main.cpp)
  target_compile_definitions(OpenDesertVR PRIVATE UNICODE _UNICODE)
  target_link_libraries(OpenDesertVR PRIVATE odv_openxr user32 gdi32 shell32 ole32)
  list(APPEND odv_targets odv_d3d12 odv_openxr odv-testhost OpenDesertVR)
  foreach(app odv-cli odv-testhost OpenDesertVR)
    add_custom_command(TARGET ${app} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:openxr_loader> $<TARGET_FILE_DIR:${app}>)
  endforeach()
  install(TARGETS odv-cli odv-testhost OpenDesertVR RUNTIME DESTINATION . COMPONENT Runtime)
  install(FILES $<TARGET_FILE:openxr_loader> DESTINATION . COMPONENT Runtime)
  install(DIRECTORY config DESTINATION . COMPONENT Runtime)
  install(FILES README.md LICENSE LICENSE-MIT THIRD_PARTY_NOTICES.md DESTINATION . COMPONENT Runtime)
  install(FILES docs/START-HERE.txt DESTINATION . COMPONENT Runtime)
  install(DIRECTORY docs DESTINATION . COMPONENT Runtime)
  install(FILES Codex_Master_Prompt.md DESTINATION . COMPONENT Runtime)
  install(FILES scripts/run-headset-test.ps1 scripts/Run-Headset-Test.cmd DESTINATION . COMPONENT Runtime)
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION .)
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_COMPONENT Runtime)
  include(InstallRequiredSystemLibraries)
endif()
foreach(target IN LISTS odv_targets)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
