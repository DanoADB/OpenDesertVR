cmake_minimum_required(VERSION 3.24)
project(OpenDesertVR VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(odv_math STATIC src/math/stereo.cpp)
target_include_directories(odv_math PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(odv_math PRIVATE /W4 /permissive- /EHsc)
else()
  target_compile_options(odv_math PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(odv_math_tests tests/math_tests.cpp)
  target_link_libraries(odv_math_tests PRIVATE odv_math)
  add_test(NAME odv_math_tests COMMAND odv_math_tests)
endif()
