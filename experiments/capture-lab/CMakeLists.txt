cmake_minimum_required(VERSION 3.24)
project(OpenDesertCaptureLab LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Capture lab requires Windows x64")
endif()
set(RENDERDOC_INCLUDE_DIR "" CACHE PATH "Directory containing the external renderdoc_app.h")
if(NOT EXISTS "${RENDERDOC_INCLUDE_DIR}/renderdoc_app.h")
  message(FATAL_ERROR "Provide -DRENDERDOC_INCLUDE_DIR from the external RenderDoc portable tool")
endif()
add_executable(odv_capture_fixture fixture.cpp)
target_compile_features(odv_capture_fixture PRIVATE cxx_std_20)
target_include_directories(odv_capture_fixture PRIVATE "${RENDERDOC_INCLUDE_DIR}")
target_link_libraries(odv_capture_fixture PRIVATE d3d12)
if(MSVC)
  target_compile_options(odv_capture_fixture PRIVATE /W4 /EHsc)
endif()
# Opt-in research fixture only. No install targets and no game dependencies.
