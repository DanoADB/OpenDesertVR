cmake_minimum_required(VERSION 3.24)
project(OpenDesertSelectedReadLab LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Selected-read lab requires Windows x64")
endif()
add_library(odv_selected_read SHARED selected_read.cpp)
target_compile_definitions(odv_selected_read PRIVATE ODV_SELECTED_READ_EXPORTS)
target_compile_features(odv_selected_read PUBLIC cxx_std_20)
add_executable(odv_selected_read_tests selected_read_tests.cpp)
target_link_libraries(odv_selected_read_tests PRIVATE odv_selected_read)
if(MSVC)
  foreach(target odv_selected_read odv_selected_read_tests)
    target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
  endforeach()
endif()
enable_testing()
add_test(NAME selected_read_owned COMMAND odv_selected_read_tests)
set_tests_properties(selected_read_owned PROPERTIES TIMEOUT 20)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME selected_read_owned_child COMMAND "${Python3_EXECUTABLE}"
  "${CMAKE_CURRENT_SOURCE_DIR}/owned_process.py" "$<TARGET_FILE:odv_selected_read>")
set_tests_properties(selected_read_owned_child PROPERTIES TIMEOUT 30)
